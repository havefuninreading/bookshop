<script>
/**
* Google Sheet 設定
* 請確保該 Google Sheet 已開啟「知道連結的任何人都能檢視」權限。
*
* 欄位對應：
* - Column A：分類（空白時自動沿用上一列分類）
* - Column B：書名
* - Column C：書籍介紹連結
* - Column D：書籍封面圖片網址（空白時自動查詢 Open Library 作為備援）
*/
const SPREADSHEET_ID = '12zw_Tm9rmINLc6dT9lll9TvsJXzu95TxymGQz9bRYUw';
const SHEET_NAME = 'SALE-LIST';
const CATEGORY_NAV_ROWS = 2;

const dynamicCoverCache = new Map();

const state = {
categories: [],
activeCategory: 'ALL',
searchQuery: '',
onlyWithLinks: false,
};

let elements = {};

function buildGvizUrl(tqxParam) {
const base = `https://docs.google.com/spreadsheets/d/${SPREADSHEET_ID}/gviz/tq`;
const params = new URLSearchParams({
tqx: tqxParam,
sheet: SHEET_NAME,
});
return `${base}?${params.toString()}`;
}

function parseGvizResponseText(rawText) {
const startIdx = rawText.indexOf('{');
const endIdx = rawText.lastIndexOf('}');
if (startIdx === -1 || endIdx === -1 || endIdx <= startIdx) {
throw new Error('無法解析 Google Sheet 回傳的資料格式。');
}
return JSON.parse(rawText.slice(startIdx, endIdx + 1));
}

function fetchViaJsonp() {
return new Promise((resolve, reject) => {
const callbackName = `gvizCallback_${Date.now()}_${Math.floor(Math.random() * 10000)}`;
const script = document.createElement('script');
const timeoutId = window.setTimeout(() => {
cleanup();
reject(new Error('讀取 Google Sheet 逾時，請檢查網路連線或試算表權限。'));
}, 12000);

function cleanup() {
window.clearTimeout(timeoutId);
delete window[callbackName];
if (script.parentNode) {
script.parentNode.removeChild(script);
}
}

window[callbackName] = (data) => {
cleanup();
resolve(data);
};

script.src = buildGvizUrl(`responseHandler:${callbackName}`);
script.onerror = () => {
cleanup();
reject(new Error('無法載入 Google Sheet 資料，請確認試算表已開啟檢視權限。'));
};

document.head.appendChild(script);
});
}

async function fetchSheetData() {
if (window.location.protocol === 'file:') {
return fetchViaJsonp();
}

try {
const response = await fetch(buildGvizUrl('out:json'), { cache: 'no-store' });
if (!response.ok) {
throw new Error(`HTTP 錯誤 (${response.status})`);
}
const text = await response.text();
return parseGvizResponseText(text);
} catch (err) {
return fetchViaJsonp();
}
}

function getCellValue(row, colIndex) {
const cell = row?.c?.[colIndex];
if (!cell || cell.v === null || cell.v === undefined) {
return '';
}
return String(cell.v).trim();
}

function normalizeUrl(rawUrl) {
if (!rawUrl) return '';
try {
const parsed = new URL(rawUrl);
if (parsed.protocol === 'http:' || parsed.protocol === 'https:') {
return parsed.href;
}
return '';
} catch {
return '';
}
}

function cleanQueryTitle(title) {
return title
.replace(/\bICR\s*:?\s*/gi, 'I Can Read ')
.replace(/\/\s*My First\b/gi, ' My First')
.replace(/\/\s*HC\b/gi, '')
.replace(/#0?(\d+)/g, ' $1 ')
.replace(/[:/]/g, ' ')
.replace(/\s+/g, ' ')
.trim();
}

async function fetchFallbackCover(title) {
const lower = title.trim().toLowerCase();
if (dynamicCoverCache.has(lower)) {
return dynamicCoverCache.get(lower);
}
dynamicCoverCache.set(lower, null);

try {
const q = encodeURIComponent(cleanQueryTitle(title));
const res = await fetch(`https://openlibrary.org/search.json?limit=1&q=${q}`);
if (!res.ok) return null;
const data = await res.json();
const coverId = data?.docs?.[0]?.cover_i;
if (coverId) {
const url = `https://covers.openlibrary.org/b/id/${coverId}-M.jpg`;
dynamicCoverCache.set(lower, url);
return url;
}
} catch {
// Ignore fallback lookup errors silently
}
return null;
}

function parseRowsToCategories(gvizData) {
if (gvizData?.status === 'error') {
const message = gvizData.errors?.[0]?.detailed_message || 'Google Sheet 回傳錯誤';
throw new Error(message);
}

const rows = gvizData?.table?.rows || [];
const categoryMap = new Map();
const categories = [];
let currentCategoryName = '未分類';

rows.forEach((row, index) => {
const colA = getCellValue(row, 0);
const colB = getCellValue(row, 1);
const colC = getCellValue(row, 2);
const colD = getCellValue(row, 3);

// 略過表頭列
if (index === 0 && (colA === '類別' || colB === '書名')) {
return;
}

// 略過完全空白列
if (!colA && !colB) {
return;
}

// Column A 有值時更新目前分類；空白時沿用上一個分類
if (colA) {
currentCategoryName = colA;
}

// 確保分類物件存在
if (!categoryMap.has(currentCategoryName)) {
const catObj = {
name: currentCategoryName,
books: [],
};
categoryMap.set(currentCategoryName, catObj);
categories.push(catObj);
}

// Column B 有書名時加入該分類清單，並讀取 Column D 封面網址
if (colB) {
const sheetCover = normalizeUrl(colD);
const cachedCover = dynamicCoverCache.get(colB.trim().toLowerCase()) || '';
categoryMap.get(currentCategoryName).books.push({
title: colB,
url: normalizeUrl(colC),
coverUrl: sheetCover || cachedCover,
});
}
});

return categories;
}

function getFilteredCategories() {
const query = state.searchQuery.trim().toLowerCase();

return state.categories
.filter((cat) => state.activeCategory === 'ALL' || cat.name === state.activeCategory)
.map((cat) => {
const categoryMatchesQuery = query && cat.name.toLowerCase().includes(query);
const filteredBooks = cat.books.filter((book) => {
if (state.onlyWithLinks && !book.url) {
return false;
}
if (!query || categoryMatchesQuery) {
return true;
}
return book.title.toLowerCase().includes(query);
});

return {
name: cat.name,
books: filteredBooks,
};
})
.filter((cat) => cat.books.length > 0);
}

function renderCategoryNav() {
const nav = elements.categoryNav;
nav.textContent = '';

const query = state.searchQuery.trim().toLowerCase();
const countBookMatch = (cat) => {
const categoryMatchesQuery = query && cat.name.toLowerCase().includes(query);
return cat.books.filter((book) => {
if (state.onlyWithLinks && !book.url) return false;
if (!query || categoryMatchesQuery) return true;
return book.title.toLowerCase().includes(query);
}).length;
};

const totalCount = state.categories.reduce((sum, cat) => sum + countBookMatch(cat), 0);
const tabs = [createCategoryTab('ALL', '全部類別', totalCount)];

state.categories.forEach((cat) => {
const count = countBookMatch(cat);
tabs.push(createCategoryTab(cat.name, cat.name, count));
});

let startIndex = 0;
for (let i = 0; i < CATEGORY_NAV_ROWS; i++) {
const size =
Math.floor(tabs.length / CATEGORY_NAV_ROWS) +
(i < tabs.length % CATEGORY_NAV_ROWS ? 1 : 0);
const rowItems = tabs.slice(startIndex, startIndex + size);
startIndex += size;

if (rowItems.length === 0) continue;

const rowDiv = document.createElement('div');
rowDiv.className = 'category-nav-row';
rowItems.forEach((tab) => rowDiv.appendChild(tab));
nav.appendChild(rowDiv);
}
}

function createCategoryTab(value, label, count) {
const btn = document.createElement('button');
btn.type = 'button';
btn.className = `category-tab${state.activeCategory === value ? ' active' : ''}`;
btn.addEventListener('click', () => {
state.activeCategory = value;
renderCategoryNav();
renderBooks();
});

const textSpan = document.createElement('span');
textSpan.textContent = label;

const countSpan = document.createElement('span');
countSpan.className = 'tab-count';
countSpan.textContent = String(count);

btn.appendChild(textSpan);
btn.appendChild(countSpan);
return btn;
}

const totalWithLinksShown = filteredCategories.reduce(
(sum, cat) => sum + cat.books.filter((b) => Boolean(b.url)).length,
0
);

elements.statsSummary.textContent = `共 ${filteredCategories.length} 個分類・${totalBooksShown} 本書（${totalWithLinksShown} 本含介紹連結）`;

const container = elements.contentArea;
container.textContent = '';

if (filteredCategories.length === 0) {
const emptyBox = document.createElement('div');
emptyBox.className = 'status-box';

const title = document.createElement('p');
title.className = 'status-title';
title.textContent = '找不到符合條件的書籍';

const desc = document.createElement('p');
desc.className = 'status-desc';
desc.textContent = '請嘗試清除搜尋關鍵字或切換其他分類。';

emptyBox.appendChild(title);
emptyBox.appendChild(desc);
container.appendChild(emptyBox);
return;
}

const listWrapper = document.createElement('div');
listWrapper.className = 'categories-list';

filteredCategories.forEach((cat) => {
const section = document.createElement('section');
section.className = 'category-section';

const header = document.createElement('div');
header.className = 'category-header';

const titleEl = document.createElement('h2');
titleEl.className = 'category-title';

const marker = document.createElement('span');
marker.className = 'category-marker';
marker.setAttribute('aria-hidden', 'true');

const titleText = document.createElement('span');
titleText.textContent = cat.name;

titleEl.appendChild(marker);
titleEl.appendChild(titleText);

const linkedCount = cat.books.filter((b) => Boolean(b.url)).length;
const metaEl = document.createElement('span');
metaEl.className = 'category-meta';
metaEl.textContent = `${cat.books.length} 本書・${linkedCount} 篇介紹`;

header.appendChild(titleEl);
header.appendChild(metaEl);

const ul = document.createElement('ul');
ul.className = 'book-list';

cat.books.forEach((book) => {
const li = document.createElement('li');
li.className = 'book-item';
li.appendChild(createBookCard(book));
ul.appendChild(li);
});

section.appendChild(header);
section.appendChild(ul);
listWrapper.appendChild(section);
});

container.appendChild(listWrapper);
}

function attachCoverImage(coverWrap, url, title) {
const img = document.createElement('img');
img.className = 'book-cover-img';
img.src = url;
img.alt = `${title} 封面`;
img.loading = 'lazy';
img.decoding = 'async';
img.referrerPolicy = 'no-referrer';

img.onerror = () => {
coverWrap.textContent = '';
coverWrap.appendChild(createPlaceholderIcon());
};

coverWrap.textContent = '';
coverWrap.appendChild(img);
}

function createBookCard(book) {
const card = book.url ? document.createElement('a') : document.createElement('div');
card.className = 'book-card';

if (book.url) {
card.href = book.url;
card.target = '_blank';
card.rel = 'noopener noreferrer';
}

const coverWrap = document.createElement('div');
coverWrap.className = 'book-cover-wrap';

if (book.coverUrl) {
attachCoverImage(coverWrap, book.coverUrl, book.title);
} else {
coverWrap.appendChild(createPlaceholderIcon());
fetchFallbackCover(book.title).then((foundUrl) => {
if (foundUrl) {
book.coverUrl = foundUrl;
attachCoverImage(coverWrap, foundUrl, book.title);
}
});
}

card.appendChild(coverWrap);

const infoDiv = document.createElement('div');
infoDiv.className = 'book-info';

const titleP = document.createElement('p');
titleP.className = 'book-title';
titleP.textContent = book.title;
infoDiv.appendChild(titleP);

card.appendChild(infoDiv);

if (book.url) {
const badge = document.createElement('span');
badge.className = 'link-badge';
badge.textContent = '閱讀介紹 ↗';
card.appendChild(badge);
} else {
const noLink = document.createElement('span');
noLink.className = 'no-link-badge';
noLink.textContent = '尚無連結';
card.appendChild(noLink);
}

return card;
}

function renderLoading() {
elements.statsSummary.textContent = '讀取中...';
elements.contentArea.innerHTML = `
<div class="status-box" role="status">
<div class="spinner" aria-hidden="true"></div>
<p class="status-title">正在讀取書單資料...</p>
<p class="status-desc">正從線上書單同步最新分類與書籍連結</p>
</div>
`;
}

function renderError(message) {
elements.statsSummary.textContent = '讀取失敗';
const box = document.createElement('div');
box.className = 'status-box error-box';

const title = document.createElement('p');
title.className = 'status-title';
title.textContent = '無法讀取 Google Sheet 資料';

const desc = document.createElement('p');
desc.className = 'status-desc';
desc.textContent = message;

box.appendChild(title);
box.appendChild(desc);

elements.contentArea.textContent = '';
elements.contentArea.appendChild(box);
}

async function init() {
renderLoading();
try {
const rawData = await fetchSheetData();
state.categories = parseRowsToCategories(rawData);

if (
state.activeCategory !== 'ALL' &&
!state.categories.some((cat) => cat.name === state.activeCategory)
) {
state.activeCategory = 'ALL';
}

renderCategoryNav();
renderBooks();
} catch (err) {
renderError(err instanceof Error ? err.message : '發生未知錯誤，請稍後再試。');
}
}

function bootstrap() {
elements = {
contentArea: document.getElementById('content-area'),
categoryNav: document.getElementById('category-nav'),
searchInput: document.getElementById('search-input'),
clearSearchBtn: document.getElementById('clear-search-btn'),
onlyLinksCheckbox: document.getElementById('only-links-checkbox'),
statsSummary: document.getElementById('stats-summary'),
};

elements.searchInput.addEventListener('input', (event) => {
state.searchQuery = event.target.value;
elements.clearSearchBtn.classList.toggle('visible', state.searchQuery.length > 0);
renderCategoryNav();
renderBooks();
});

elements.clearSearchBtn.addEventListener('click', () => {
state.searchQuery = '';
elements.searchInput.value = '';
elements.clearSearchBtn.classList.remove('visible');
elements.searchInput.focus();
renderCategoryNav();
renderBooks();
});

elements.onlyLinksCheckbox.addEventListener('change', (event) => {
state.onlyWithLinks = event.target.checked;
renderCategoryNav();
renderBooks();
});

init();
}

if (document.readyState === 'loading') {
document.addEventListener('DOMContentLoaded', bootstrap);
} else {
bootstrap();
}
